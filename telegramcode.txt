void setup() {

  pinMode(LDRPIN, INPUT);  // Set LDR pin as input

void loop() {
  delay(2000);
  
  float tc = dht.readTemperature(false);  // Read temperature in C
  float tf = dht.readTemperature(true);   // Read Temperature in F
  float hu = dht.readHumidity();          // Read Humidity

  int soilMoisture = analogRead(soil_moisture_pin);  // Read soil moisture
  int ldrValue = analogRead(LDRPIN);    // Read LDR value
  int ldrDigitalValue = digitalRead(LDR_DIGITAL_PIN); // Read LDR digital value
 

  // Print the sensor values to the Serial Monitor
  Serial.print("Temp: ");
  Serial.print(tc);
  Serial.print(" C, ");
  Serial.print(tf);
  Serial.print(" F, Hum: ");
  Serial.print(hu);
  Serial.println("%");

  Serial.print("Soil Moisture: ");
  Serial.println(soilMoisture);

   Serial.print("LDR Analog Value: ");
  Serial.println(ldrValue); // when it equal 0 the light very high , when equal approximatly 3500 it dark


  Serial.print("LDR Digital Value: ");
  if (ldrDigitalValue == HIGH) {
    Serial.println("LED OFF");
  } else {
    Serial.println("LED ON");
  }
  // Prepare message to send
  String message = "Sensor Data:\n";
  message += "Temperature: " + String(tc) + " °C\n";
  message += "Humidity: " + String(hu) + " %\n";
  message += "Soil Moisture: " + String(soilMoisture) + "\n";
  message += "LDR Analog Value: " + String(ldrValue) + "\n";
  message += "LDR Digital Value: " + String(ldrDigitalValue == HIGH ? "LED OFF" : "LED ON") + "\n";


  // Manually URL encode the message (replace spaces with %20, etc.)
  String encodedMessage = encodeURL(message);

  // Send message to Telegram with error handling and retry
  if (sendTelegramMessage(encodedMessage)) {
    Serial.println("Message sent successfully");
  } else {
    Serial.println("Error sending message, retrying...");
    delay(1000);  // Wait before retrying
    if (sendTelegramMessage(encodedMessage)) {
      Serial.println("Message sent successfully after retry");
    } else {
      Serial.println("Failed to send message after retry");
    }
  }

  // Delay to avoid spamming Telegram
  delay(20000);
}

bool sendTelegramMessage(String message) {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    String url = String("https://api.telegram.org/bot") + botToken + "/sendMessage?chat_id=" + chatID + "&text=" + message;

    // Debug the full URL
    Serial.println("URL: " + url);

    http.begin(url);
    http.setTimeout(10000);  // Set a timeout of 10 seconds
    int httpCode = http.GET();
    
    if (httpCode > 0) {
      http.end();  // Close connection after each request
      return true; // Message sent successfully
    } else {
      http.end();  // Close connection if error occurred
      Serial.println("HTTP request failed, code: " + String(httpCode));
      return false; // Error sending message
    }
  } else {
    Serial.println("WiFi not connected");
    return false;
  }
}

String encodeURL(String str) {
  String encoded = "";
  char c;
  for (unsigned int i = 0; i < str.length(); i++) {
    c = str.charAt(i);
    if (c == ' ') {
      encoded += "%20";
    } else if (c == '\n') {
      encoded += "%0A";
    } else if (c == '\r') {
      encoded += "%0D";
    } else if (c == '!') {
      encoded += "%21";
    } else if (c == '"') {
      encoded += "%22";
    } else if (c == '#') {
      encoded += "%23";
    } else if (c == '$') {
      encoded += "%24";
    } else if (c == '%') {
      encoded += "%25";
    } else if (c == '&') {
      encoded += "%26";
    } else if (c == '\'') {
      encoded += "%27";
    } else if (c == '(') {
      encoded += "%28";
    } else if (c == ')') {
      encoded += "%29";
    } else if (c == '*') {
      encoded += "%2A";
    } else if (c == '+') {
      encoded += "%2B";
    } else if (c == ',') {
      encoded += "%2C";
    } else if (c == '-') {
      encoded += "%2D";
    } else if (c == '.') {
      encoded += "%2E";
    } else if (c == '/') {
      encoded += "%2F";
    } else {
      encoded += c;
    }
  }
  return encoded;
}
